## Scope

### Настройка Playwright

Установить в `client/`:
```bash
npm install -D @playwright/test
npx playwright install chromium
```

Создать `client/playwright.config.ts`:
```ts
import { defineConfig } from '@playwright/test';
export default defineConfig({
  testDir: './e2e',
  use: {
    baseURL: process.env.BASE_URL ?? 'http://localhost:8080',
  },
  webServer: {                       // опционально: запуск dev-сервера
    command: 'npm run dev',
    port: 5173,
    reuseExistingServer: true,
  },
});
```

Добавить скрипт в `client/package.json`:
```json
"test:e2e": "playwright test"
```

### Тест: `client/e2e/arena.spec.ts`

```ts
import { test, expect } from '@playwright/test';

test('two players join → log shows both joined', async ({ browser }) => {
  // Шаг 1: Создатель — открыть лендинг, создать комнату
  const ctx1 = await browser.newContext();
  const page1 = await ctx1.newPage();
  await page1.goto('/');
  await page1.getByRole('button', { name: /создать/i }).click();
  // Ждём редиректа на /g/{code}
  await page1.waitForURL(/\/g\//);
  const arenaUrl = page1.url();

  // Шаг 2: Создатель вводит имя
  await page1.getByPlaceholder(/имя/i).fill('Создатель');
  await page1.getByRole('button', { name: /войти/i }).click();
  // Ждём появления canvas (Phaser)
  await expect(page1.locator('canvas')).toBeVisible({ timeout: 10_000 });

  // Шаг 3: Гость — открыть тот же URL
  const ctx2 = await browser.newContext();
  const page2 = await ctx2.newPage();
  await page2.goto(arenaUrl);
  await page2.getByPlaceholder(/имя/i).fill('Гость');
  await page2.getByRole('button', { name: /войти/i }).click();
  await expect(page2.locator('canvas')).toBeVisible({ timeout: 10_000 });

  // Шаг 4: Оба в логе видят player_joined
  // (лог рендерится в canvas — проверяем косвенно через WS-событие или data-testid)
  // Если лог сделан через DOM overlay: использовать page.locator('[data-testid="log"]')
  // Если только canvas: достаточно проверить, что оба canvas видны без ошибок консоли
  const errors1: string[] = [];
  page1.on('console', m => m.type() === 'error' && errors1.push(m.text()));
  await page1.waitForTimeout(2000);
  expect(errors1).toHaveLength(0);

  await ctx1.close();
  await ctx2.close();
});

test('non-existent code → no crash', async ({ page }) => {
  await page.goto('/g/NONEXISTENT');
  // Арена открывается, но WS закрывается с ошибкой
  await page.getByPlaceholder(/имя/i).fill('Test');
  await page.getByRole('button', { name: /войти/i }).click();
  // Ожидаем graceful error (зависит от реализации E3/E4)
  await page.waitForTimeout(2000);
  // Не должно быть uncaught exception
});
```

### CI (опционально)

Добавить в `.github/workflows/ci.yml` job `e2e`:
```yaml
e2e:
  runs-on: ubuntu-latest
  steps:
    - uses: actions/checkout@v4
    - uses: actions/setup-node@v4
      with: { node-version: '20' }
    - run: npm ci
      working-directory: client
    - run: npx playwright install --with-deps chromium
      working-directory: client
    - run: docker compose up -d --build
    - run: npm run test:e2e
      working-directory: client
      env:
        BASE_URL: http://localhost:8080
    - run: docker compose down
```
